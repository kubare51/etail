# ---------------------------------------------------------
# ENVIRONMENT VARIABLES FOR WEBHOOK
# Replace placeholder values before deploying
# ---------------------------------------------------------
SetEnv WEBHOOK_SECRET "YOUR_WEBHOOK_SECRET_HERE"
SetEnv REPO_PATH "/home/USERNAME/public_html/CLIENT_FOLDER"
SetEnv DEPLOY_SCRIPT "/home/USERNAME/public_html/CLIENT_FOLDER/deploy.sh"
SetEnv DEPLOY_BRANCH "staging"
SetEnv LOG_FILE "/home/USERNAME/logs/deploy.log"
SetEnv ENVIRONMENT "staging"

# GitHub Token (optional - for notifications)
# SetEnv GITHUB_TOKEN "ghp_xxxxxxxxxxxx"

# Dashboard API URL (for logging deployments)
SetEnv DASHBOARD_API_URL "https://retail.logimaxindia.com/retail_v5/cicd/docs/api"

# ---------------------------------------------------------
# PHP SETTINGS
# ---------------------------------------------------------
<IfModule mod_php7.c>
    php_value max_execution_time 300
    php_value memory_limit 256M
    php_flag display_errors Off
    php_flag log_errors On
</IfModule>

<IfModule mod_php8.c>
    php_value max_execution_time 300
    php_value memory_limit 256M
    php_flag display_errors Off
    php_flag log_errors On
</IfModule>

# ---------------------------------------------------------
# CORS HEADERS FOR DASHBOARD API
# ---------------------------------------------------------
<IfModule mod_headers.c>
    Header always set Access-Control-Allow-Origin "*"
    Header always set Access-Control-Allow-Methods "GET, POST, OPTIONS"
    Header always set Access-Control-Allow-Headers "Content-Type, X-Hub-Signature-256, X-GitHub-Event"
    
    # Handle preflight requests
    RewriteEngine On
    RewriteCond %{REQUEST_METHOD} OPTIONS
    RewriteRule ^(.*)$ $1 [R=200,L]
</IfModule>

# ---------------------------------------------------------
# SECURITY RULES
# ---------------------------------------------------------

# Deny access to sensitive files
<FilesMatch "\.(sh|log|env|git|sql|md)$">
    Order Allow,Deny
    Deny from all
</FilesMatch>

# Protect .htaccess itself
<Files .htaccess>
    Order Allow,Deny
    Deny from all
</Files>

# Allow only POST to deploy.php
<Files "deploy.php">
    <LimitExcept POST>
        Order Deny,Allow
        Deny from all
    </LimitExcept>
</Files>

# Block common attack patterns
<IfModule mod_rewrite.c>
    RewriteEngine On
    
    # Block suspicious query strings
    RewriteCond %{QUERY_STRING} (\<|%3C).*script.*(\>|%3E) [NC,OR]
    RewriteCond %{QUERY_STRING} GLOBALS(=|\[|\%[0-9A-Z]{0,2}) [OR]
    RewriteCond %{QUERY_STRING} _REQUEST(=|\[|\%[0-9A-Z]{0,2})
    RewriteRule .* - [F,L]
    
    # Block bad bots
    RewriteCond %{HTTP_USER_AGENT} ^$ [OR]
    RewriteCond %{HTTP_USER_AGENT} ^(java|curl|wget).* [NC]
    RewriteCond %{HTTP_USER_AGENT} !^curl.* [NC]
    RewriteRule .* - [F,L]
</IfModule>
