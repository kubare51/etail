INSERT INTO ret_import_tag_details (
    old_metal_id, metal_id, metal_name,
    old_category_id, category_id, category_name,
    old_product_id, product_id, product_name,
    old_design_id, design_id, design_name,
    old_sub_design_id, sub_design_id, sub_design_name,
    old_purity_id, purity_id, purity_name,
    old_branch_id, branch_id, branch_name,
    old_karigar_id, karigar_id, karigar_name,
    old_counter_id, counter_id, counter_name,
    old_size_id, size_id, size_value, size_name,

    pieces, gross_wt, net_wt, less_wt,
    sales_type, prod_calc_based_on, tag_calc_based_on,
    wastage_per, wastage_wt,
    mc_type, mc_value,
    sales_value, rate_per_gram, sell_rate,

    mfr_code, style_code, tag_number,
    certification_no, purchase_cost,

    tag_date, huid1, huid2,

    updatestatus, created_on, updated_on
)
SELECT 
  IFNULL(t.tag_id, ''),
  NULL, IFNULL(m.metal, ''),

  IFNULL(cat.id_ret_category, ''), NULL, IFNULL(cat.name, ''),

  IFNULL(pro.pro_id, ''), NULL, IFNULL(pro.product_name, ''),

  IFNULL(des.design_no, ''), NULL, IFNULL(des.design_name, ''),

  IFNULL(des.design_no, ''), NULL, IFNULL(des.design_name, ''),

  IFNULL(p.id_purity, ''), NULL, IFNULL(p.purity, ''),

  IFNULL(b.id_branch, ''), NULL, IFNULL(b.name, ''),

  IFNULL(kar.id_karigar, ''), NULL, IFNULL(kar.firstname, ''),

  IFNULL(sec.id_section, ''), NULL, IFNULL(sec.section_name, ''),

  IFNULL(s.id_size, ''), NULL, IFNULL(s.value, ''), IFNULL(s.name, ''),

  IFNULL(t.piece, 0),
  IFNULL(t.gross_wt, 0.000),
  IFNULL(t.net_wt, 0.000),
  IFNULL(t.less_wt, 0.000),

  IFNULL(pro.sales_mode, ''),
  IFNULL(pro.calculation_based_on, ''),
  IFNULL(t.calculation_based_on, ''),

  IFNULL(t.retail_max_wastage_percent, 0.000),
  ROUND(IFNULL(t.net_wt, 0.000) * IFNULL(t.retail_max_wastage_percent, 0.000) / 100.0, 3),

  IFNULL(t.tag_mc_type, ''),
  IFNULL(t.tag_mc_value, 0.00),

  IFNULL(t.sales_value, 0.00),
  IFNULL(t.item_rate, 0.00),
  IFNULL(t.sell_rate, 0.00),

  NULL, NULL, IFNULL(t.tag_code, ''),
  NULL, NULL,

  IFNULL(t.tag_datetime, ''),
  IFNULL(t.hu_id, ''), NULL,

  '0', NOW(), NULL

FROM ret_taging t 
LEFT JOIN ret_product_master pro ON pro.pro_id = t.product_id
LEFT JOIN ret_design_master des ON des.design_no = t.design_id
LEFT JOIN ret_category cat ON cat.id_ret_category = pro.cat_id
LEFT JOIN metal m ON m.id_metal = cat.id_metal
LEFT JOIN ret_purity p ON p.id_purity = t.purity
LEFT JOIN branch b ON b.id_branch = t.current_branch
LEFT JOIN ret_lot_inwards l ON l.lot_no = t.tag_lot_id
LEFT JOIN ret_karigar kar ON kar.id_karigar = l.gold_smith
LEFT JOIN ret_section sec ON sec.id_section = t.counter
LEFT JOIN ret_size s ON s.id_size = t.size
WHERE t.tag_id IS NOT NULL 
  AND t.tag_status = 0
  AND cat.id_ret_category IS NOT NULL 
  AND pro.pro_id IS NOT NULL;

INSERT INTO ret_import_stone_details (
    tag_id, tag_number, is_less_id,
    old_stone_type_id, stone_type_id, stone_type_name,
    old_stone_id, stone_id, stone_name,
    old_uom_id, uom_id, uom_name,
    pieces, stone_wt, stone_calc_type, rate_per_gram, amount,
    updatestatus, created_on
)
SELECT 
  IFNULL(tst.tag_id, '') AS tag_id,
  IFNULL(t.tag_code, '') AS tag_number,
  IFNULL(tst.is_apply_in_lwt, 0) AS is_less_id,

  IFNULL(rst.id_stone_type, '') AS old_stone_type_id,
  NULL AS stone_type_id,
  IFNULL(rst.stone_type, '') AS stone_type_name,

  IFNULL(st.stone_id, '') AS old_stone_id,
  NULL AS stone_id,
  IFNULL(st.stone_name, '') AS stone_name,

  IFNULL(u.uom_id, '') AS old_uom_id,
  NULL AS uom_id,
  IFNULL(u.uom_name, '') AS uom_name,

  IFNULL(tst.pieces, 0) AS pieces,
  IFNULL(tst.wt, 0.000) AS stone_wt,
  IFNULL(tst.stone_cal_type, '') AS stone_calc_type,
  IFNULL(tst.rate_per_gram, 0.000) AS rate_per_gram,
  IFNULL(tst.amount, 0.000) AS amount,

  '0' AS updatestatus,
  NOW() AS created_on

FROM ret_taging_stone tst
LEFT JOIN ret_taging t ON t.tag_id = tst.tag_id
LEFT JOIN ret_stone st ON st.stone_id = tst.stone_id
LEFT JOIN ret_stone_type rst ON rst.id_stone_type = st.stone_type
LEFT JOIN ret_uom u ON u.uom_id = tst.uom_id;

SET td.metal_id = m.id_metal;

/* -------------------- CATEGORY -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_category c 
    ON UPPER(TRIM(c.name)) = UPPER(TRIM(td.category_name))
SET td.category_id = c.id_ret_category;

/* -------------------- PRODUCT -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_product_master p 
    ON UPPER(TRIM(p.product_name)) = UPPER(TRIM(td.product_name))
SET td.product_id = p.pro_id
WHERE td.product_id is NULL;

/* -------------------- DESIGN -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_design_master d 
    ON UPPER(TRIM(d.design_name)) = UPPER(TRIM(td.design_name))
SET td.design_id = d.design_no
WHERE td.design_id IS NULL;

/* -------------------- SUB DESIGN -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_sub_design_master sd 
    ON UPPER(TRIM(sd.sub_design_name)) = UPPER(TRIM(td.sub_design_name))
SET td.sub_design_id = sd.id_sub_design;

/* -------------------- PURITY -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_purity pu 
    ON UPPER(TRIM(pu.purity)) = UPPER(TRIM(td.purity_name))
SET td.purity_id = pu.id_purity;

/* -------------------- BRANCH -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN branch b 
    ON UPPER(TRIM(b.name)) = UPPER(TRIM(td.branch_name))
SET td.branch_id = b.id_branch;

/* -------------------- KARIGAR -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_karigar k 
    ON UPPER(TRIM(k.firstname)) = UPPER(TRIM(td.karigar_name))
SET td.karigar_id = k.id_karigar;

/* -------------------- COUNTER / SECTION -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_section sec 
    ON UPPER(TRIM(sec.section_name)) = UPPER(TRIM(td.counter_name))
SET td.counter_id = sec.id_section;

/* -------------------- SIZE (depends on product_id) -------------------- */
UPDATE ret_import_tag_details td
INNER JOIN ret_size si 
    ON UPPER(TRIM(si.value)) = UPPER(TRIM(td.size_value))
   AND td.product_id = si.id_product
SET td.size_id = si.id_size;

UPDATE ret_import_tag_details t
JOIN ret_product_master p ON p.pro_id = t.product_id
SET t.product_short_code = p.product_short_code
WHERE t.product_short_code IS NULL OR t.product_short_code = '';



size var -  Illegal mix of collations (utf8mb4_unicode_ci,IMPLICIT) and (utf8mb4_0900_ai_ci,IMPLICIT) for operation '=' - for size comparision

UPDATE ret_import_tag_details td
INNER JOIN ret_size si 
    ON COALESCE(UPPER(TRIM(si.value)), '') COLLATE utf8mb4_unicode_ci
       = COALESCE(UPPER(TRIM(td.size_value)), '') COLLATE utf8mb4_unicode_ci
   AND td.product_id = si.id_product
SET td.size_id = si.id_size;

CREATE TABLE ret_import_charges (

);
